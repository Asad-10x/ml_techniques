def predict():
    logger.debug("Received request data: %s", request.data)

    if not request.json or 'data' not in request.json:
        return jsonify({'error': 'Invalid input format, expected JSON with key "data".'}), 400

    input_data = request.json['data']
    if not isinstance(input_data, dict):
        return jsonify({'error': 'Input data must be a dictionary of features.'}), 400

    try:
        # Build raw feature dict
        raw = {
            'age': safe_float(input_data.get('age')),  
            'educational-num': safe_int(input_data.get('educational_num')),
            'hours-per-week': safe_float(input_data.get('hours-per-week')),
            'capital-gain': safe_float(input_data.get('capital-gain')),
            'capital-loss': safe_float(input_data.get('capital-loss')),
            'gender': 1 if str(input_data.get('gender','')).lower() == 'male' else 0
        }

        # One-hot flags
        field_map = {
            'workclass': 'workclass_',
            'marital_status': 'marital-status_',
            'relationship_type': 'relationship_',
            'occupation_type': 'occupation_',
            'race': 'race_',
            'native_country': 'native-country_'
        }
        for field, prefix in field_map.items():
            raw[f"{prefix}{input_data.get(field)}"] = 1

        # Initialize DataFrame
        df_input = pd.DataFrame([{col: 0 for col in model_columns}])
        for k, v in raw.items():
            if k in df_input.columns:
                df_input.at[0, k] = v

        # Scale numeric features together

        print(f"Model prediction: {prediction} -> {label}")

        return jsonify({'prediction': int(prediction), 'label': label})
